```bash

	dataset \
	collect \
	[count=<count>,~download,upload] \
	[-|<object-name>]

@swallow \
	dataset \
	edit \
	[~download]
 . edit the swallow dataset.
@swallow \
	dataset \
	upload \
	[~metadata]
```
